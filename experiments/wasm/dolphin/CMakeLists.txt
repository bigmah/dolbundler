# Dolphin, ModernGekko and a statically recompiled game module, compiled to
# WebAssembly.
#
# This is the wasm sibling of ios/CMakeLists.txt and deliberately mirrors it:
# both targets forbid a JIT, so both link the recompiled module into the binary
# ahead of time and hand the runtime an AttachedDescriptor. Nothing here is
# emscripten-specific except the link options.
#
# Configure through experiments/wasm/dolphin/build.sh, which supplies emcmake.

cmake_minimum_required(VERSION 3.20)
project(DolphinWasm C CXX)

if(NOT EMSCRIPTEN)
  message(FATAL_ERROR
    "This project only builds for WebAssembly. Configure it with emcmake.")
endif()

if(NOT ENABLE_GENERIC)
  # Same reason as iOS: ENABLE_GENERIC is what keeps JitArm64/Jit64 out of the
  # build. There is no wasm JIT backend in Dolphin, so without this the
  # configure picks a host JIT and fails much later and much less clearly.
  message(FATAL_ERROR
    "ENABLE_GENERIC must be ON: it is what excludes Dolphin's JIT backends.")
endif()

set(MODERNGEKKO_ENABLE_DOLPHIN_RUNTIME ON CACHE BOOL "" FORCE)
# No SDL3 window, no imgui launcher, no port tool: the page is the frontend.
set(MODERNGEKKO_ENABLE_FRONTEND OFF CACHE BOOL "" FORCE)

add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../../../ModernGekko
                 ${CMAKE_BINARY_DIR}/moderngekko EXCLUDE_FROM_ALL)

set(DOLPHIN_CORE_DIR
    ${CMAKE_CURRENT_SOURCE_DIR}/../../../ModernGekko/vendor/dolphin/Source/Core)

# A non-void function whose body vanishes into an #if with no branch taken
# still compiles: it falls off its end, which is undefined behaviour and in
# wasm a trap rather than a wrong answer. The iOS port hit this and so does
# this one, so make it an error here too.
add_compile_options(-Werror=return-type)

add_executable(dolweb
  main.cpp
  platform_stubs.cpp
)

# Dolphin's headers branch on these and this target is outside the subdirectory
# where they are set. Without _M_GENERIC, MachineContext.h reaches
# "#error No context definition for OS" -- it has a generic arm precisely
# because a JIT-less build has no machine context to define.
target_compile_definitions(dolweb PRIVATE _M_GENERIC=1 _ARCH_32=1)

target_include_directories(dolweb PRIVATE
  ${DOLPHIN_CORE_DIR}
  ${CMAKE_CURRENT_SOURCE_DIR}/../../../ModernGekko/include
)

target_link_libraries(dolweb PRIVATE
  ModernGekko::Runtime
  moderngekko_dolvm
  fmt::fmt
)

# --- the statically recompiled game module ----------------------------------
#
# Lifted from ios/CMakeLists.txt, which solved the same problem for the same
# reason: the module cannot be dlopen'd, so it is linked in and reached through
# a descriptor. The registry keys on disc ID so one binary can carry several.
set(DOLWEB_NATIVE_MODULES "" CACHE STRING
    "Semicolon-separated GAME_ID=/generated/dir recompiled modules to embed")

if(DOLWEB_NATIVE_MODULES)
  string(CONCAT _native_registry
    "// Generated by experiments/wasm/dolphin/CMakeLists.txt; do not edit.\n"
    "struct DolWebNativeModuleEntry {\n"
    "  const char* game_id;\n"
    "  const ModernGekkoModuleDesc* (*get_module)(void);\n"
    "};\n")
  set(_native_registry_declarations "")
  set(_native_registry_entries "")
  set(_native_game_ids "")

  foreach(_native_spec IN LISTS DOLWEB_NATIVE_MODULES)
    if(NOT _native_spec MATCHES
       "^([A-Z0-9][A-Z0-9][A-Z0-9][A-Z0-9][A-Z0-9][A-Z0-9])=(.+)$")
      message(FATAL_ERROR
        "Invalid native module '${_native_spec}'; expected GAME_ID=/generated/dir")
    endif()
    set(_native_game_id "${CMAKE_MATCH_1}")
    set(_native_generated_dir "${CMAKE_MATCH_2}")
    if(_native_game_id IN_LIST _native_game_ids)
      message(FATAL_ERROR "Duplicate native module game ID: ${_native_game_id}")
    endif()
    if(NOT EXISTS "${_native_generated_dir}/generated.h")
      message(FATAL_ERROR
        "Native module ${_native_game_id} is missing ${_native_generated_dir}/generated.h")
    endif()

    file(READ "${_native_generated_dir}/generated.h" _native_header)
    set(_native_symbol_prefix "")
    string(REGEX MATCH
      "#define DOLRECOMP_NATIVE_SYMBOL_PREFIX \"[^\"]*\""
      _native_symbol_prefix_line "${_native_header}")
    if(_native_symbol_prefix_line)
      string(REGEX REPLACE
        "^#define DOLRECOMP_NATIVE_SYMBOL_PREFIX \"([^\"]*)\"$" "\\1"
        _native_symbol_prefix "${_native_symbol_prefix_line}")
    endif()
    if(_native_symbol_prefix AND
       NOT _native_symbol_prefix MATCHES "^[A-Za-z_][A-Za-z0-9_]*$")
      message(FATAL_ERROR
        "Invalid symbol prefix for ${_native_game_id}: ${_native_symbol_prefix}")
    endif()
    set(_native_getter "${_native_symbol_prefix}staticrecomp_get_module")

    list(APPEND _native_game_ids "${_native_game_id}")
    string(APPEND _native_registry_declarations
      "extern \"C\" const ModernGekkoModuleDesc* ${_native_getter}(void);\n")
    string(APPEND _native_registry_entries
      "  {\"${_native_game_id}\", ${_native_getter}},\n")

    set(GAME_ID "${_native_game_id}")
    set(GENERATED_DIR "${_native_generated_dir}")
    set(GXRUNTIME_DIR
      "${CMAKE_CURRENT_SOURCE_DIR}/../../../ModernGekko/vendor/dolphin/GXRuntime")
    set(CHASSIS_ABI_DIR "${DOLPHIN_CORE_DIR}/Core/PowerPC/StaticRecomp")
    set(RECOMPCORE_MODULE_EMBEDDED ON)
    if(NOT DEFINED RECOMPCORE_MODULE_OPT_LEVEL)
      set(RECOMPCORE_MODULE_OPT_LEVEL 3)
    endif()
    add_subdirectory(
      "${CMAKE_CURRENT_SOURCE_DIR}/../../../ModernGekko/vendor/dolphin/module-template"
      "${CMAKE_BINARY_DIR}/native-game-${_native_game_id}")

    target_link_libraries(dolweb PRIVATE "g${_native_game_id}_recomp")
  endforeach()

  string(APPEND _native_registry "${_native_registry_declarations}\n"
    "static const DolWebNativeModuleEntry kDolWebNativeModules[] = {\n"
    "${_native_registry_entries}};\n")
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/dolweb_native_modules.inc"
    "${_native_registry}")
  target_include_directories(dolweb PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
  target_compile_definitions(dolweb PRIVATE DOLWEB_HAVE_NATIVE_MODULES=1)
endif()

# --- emscripten link ---------------------------------------------------------
#
# PROXY_TO_PTHREAD is not a tuning choice: Dolphin's core thread blocks, and a
# browser's main thread may not. It moves main() onto a worker so Run() can
# block the way it does everywhere else.
# Dolphin's arenas want ~100 MB before a disc is read and the texture cache
# grows from there. Start small and grow: a phone's WebContent process is the
# tightest budget this has to fit in, and reserving 1.5 GB up front is the
# fastest way to be killed before the first frame.
set(DOLWEB_TOTAL_MEMORY "512MB" CACHE STRING "Initial linear memory")
set(DOLWEB_MAX_MEMORY "2048MB" CACHE STRING "Ceiling for linear-memory growth")

# Binaryen's -O3 pass over a whole recompiled game is the single slowest step in
# this build and it changes nothing measurable: -O1 and -O3 links timed the same
# on the interpreter build. Every translation unit is already compiled at -O3 by
# LLVM, which is where the optimisation actually happens.
option(DOLWEB_FAST_LINK "Skip Binaryen's heavy link-time pass" ON)
if(DOLWEB_FAST_LINK)
  target_link_options(dolweb PRIVATE -O1)
endif()

target_link_options(dolweb PRIVATE
  -pthread
  -sPROXY_TO_PTHREAD
  -sPTHREAD_POOL_SIZE=16
  -sALLOW_MEMORY_GROWTH=1
  -sINITIAL_MEMORY=${DOLWEB_TOTAL_MEMORY}
  -sMAXIMUM_MEMORY=${DOLWEB_MAX_MEMORY}
  -sSTACK_SIZE=8MB
  -sASSERTIONS=0
  # A whole recompiled game is tens of megabytes of code; the default table and
  # export handling copes, but the linker needs room for the chunk functions.
  -sALLOW_TABLE_GROWTH=1
  # WebGL2 for both builds, not only the browser one. The measurement build has
  # no canvas and never draws, but it links Dolphin's OpenGL backend and the GL
  # context that times it, and those reference GLES3 entry points that only
  # exist in the GL library when WebGL2 is on.
  -sMIN_WEBGL_VERSION=2
  -sMAX_WEBGL_VERSION=2
  -sFULL_ES3=1
  -sGL_ENABLE_GET_PROC_ADDRESS=1
)

if(DOLWEB_NODE)
  # The measurement build: no canvas, no fetch, the disc read straight off the
  # host filesystem so the harness measures the emulator and not the loader.
  target_link_options(dolweb PRIVATE
    -sNODERAWFS=1 -sENVIRONMENT=node,worker
    # The harness is a command: it has to run atexit handlers and stop.
    -sEXIT_RUNTIME=1)
else()
  target_compile_definitions(dolweb PRIVATE DOLWEB_WEB=1)
  # A page is not a command. Tearing the runtime down when main() returns runs
  # Dolphin's exit handlers on a thread that is already unwinding, and the first
  # thing that goes wrong there is an indirect call through a freed table entry:
  # "function signature mismatch", after the run, saying nothing about the run.
  target_link_options(dolweb PRIVATE -sEXIT_RUNTIME=0)
  # Dolphin's Sys tree, preloaded rather than fetched. It is small once the Wii
  # title databases and the seven megabytes of per-title INIs for games this
  # build will never see are left out (stage-sys.sh), and it is read early and
  # often enough that range requests would be the wrong shape. Sys/GC carries
  # the DSP ROMs and the IPL fonts -- a game that draws its menus with the
  # system font shows an empty menu without them.
  set(DOLWEB_SYS_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../build-wasm/sys"
      CACHE PATH "Staged Dolphin Sys tree to preload into the page")
  if(EXISTS "${DOLWEB_SYS_DIR}")
    target_link_options(dolweb PRIVATE
      "SHELL:--preload-file ${DOLWEB_SYS_DIR}@/Sys")
  else()
    message(WARNING
      "No staged Sys tree at ${DOLWEB_SYS_DIR}; run stage-sys.sh first")
  endif()
  # Handing the canvas to the thread that renders makes every GL call local
  # instead of a queued cross-thread call, and on a Mac that is worth about 14%.
  #
  # It is on, but only because Core::Init now survives it not working. Handing a
  # canvas over makes pthread_create asynchronous -- the browser's main thread
  # performs the transfer and only then spawns the worker -- and on iOS that
  # sometimes never happens, with pthread_create having already returned
  # success. The emulation thread never runs, Dolphin prints nothing at all, and
  # the CPU sits in PowerDown with no error anywhere. Core::Init waits to see the
  # thread claim its work and starts a plain one if it does not, which falls back
  # to a proxied context: slower, and it starts every time.
  option(DOLWEB_OFFSCREEN_CANVAS
         "Hand the canvas to the rendering thread (faster where it lands)" ON)
  if(DOLWEB_OFFSCREEN_CANVAS)
    target_link_options(dolweb PRIVATE
      -sOFFSCREENCANVAS_SUPPORT=1
      # Quoted: an unquoted # starts a CMake comment and the setting reaches
      # emscripten empty.
      "-sOFFSCREENCANVASES_TO_PTHREAD=#canvas,#dolweb-probe")
  endif()

  # A race-free replacement for emscripten's WASMFS fetch backend. The stock one
  # creates a file's range table after an awaited HEAD, so two reads that arrive
  # before either HEAD resolves both create it and the second discards the first
  # one's chunks; the first then throws out of its async read, which reaches C++
  # as a read that moved no bytes. Dolphin reads the disc from more than one
  # thread. See dolweb-fetch.js.
  target_link_options(dolweb PRIVATE
    "SHELL:--js-library ${CMAKE_CURRENT_SOURCE_DIR}/dolweb-fetch.js")

  target_link_options(dolweb PRIVATE
    # WASMFS rather than the JS filesystem: its state lives in linear memory, so
    # every thread sees the same tree. The JS one does not -- it is per-worker
    # JS state, and Dolphin reads the disc from a thread it created itself.
    -sWASMFS=1
    -sENVIRONMENT=web,worker
    -sMODULARIZE=1
    -sEXPORT_ES6=0
    -sEXPORT_NAME=createDolWeb
    -sEXPORTED_RUNTIME_METHODS=[ccall,cwrap,HEAPU8,HEAPU32]
    -sOFFSCREEN_FRAMEBUFFER=1
  )
endif()
